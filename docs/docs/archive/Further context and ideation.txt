    public function check($agent, $action)

🔑 Key Service: FlowExecutionService
class FlowExecutionService
{
    public function execute($command, $payload)
    {
        $plan = app(MetaPlannerService::class)->plan($command, $payload);

        foreach ($plan['tasks'] as $task) {
            ExecuteAgentJob::dispatch($task)->onQueue($task['queue']);
        }

        return $plan;
    }
}

Example: Interview Agent (Python)
class BusinessInterviewAgent:
    QUESTIONS = [
        "What does your business do?",
        "What are your weekly recurring tasks?",
        "How do you get customers?",
        "What tools do you use daily?",
        "What takes the most time?"
    ]

    def run(self):
        answers = {}

        for q in self.QUESTIONS:
            answer = input(q + "\n> ")
            answers[q] = answer

        self.save_context(answers)
        return answers

    def save_context(self, answers):
        with open("workspace/profile/company.md", "w") as f:
            for q, a in answers.items():
                f.write(f"## {q}\n{a}\n\n")
________________________________________
Why this matters (theory)
•	This creates structured context grounding
•	Equivalent to building a vector database—but human-readable
•	Aligns with RAGFlow principles: retrieval > generation

apps/inference-plane/router.py
import requests
import os

LOCAL_MODEL_URL = "http://localhost:11434/api/generate"  # Ollama
OPENAI_API_KEY = os.getenv("OPENAI_API_KEY")

def generate(prompt, complexity="low"):
    try:
        # Try LOCAL first
        response = requests.post(LOCAL_MODEL_URL, json={
            "model": "gemma:2b",
            "prompt": prompt,
            "stream": False
        })

        if response.status_code == 200:
            return response.json()["response"]

    except Exception:
        pass

    # Fallback to API
    return openai_fallback(prompt)


def openai_fallback(prompt):
    r = requests.post("https://api.openai.com/v1/chat/completions",
        headers={
            "Authorization": f"Bearer {OPENAI_API_KEY}"
        },
        json={
            "model": "gpt-4o-mini",
            "messages": [{"role": "user", "content": prompt}]
        }
    )

    return r.json()["choices"][0]["message"]["content"]

def send_message(chat_id, text):
    requests.post(f"{BASE_URL}/sendMessage", json={

def handle_update(update):
    message = update["message"]["text"]
    chat_id = update["message"]["chat"]["id"]

    if message.startswith("/explore"):
        requests.post("http://control:8000/api/command", json={
            "command": "explore"

        send_message(chat_id, "Exploring your business...")

            "command": "analyze"

    elif message.startswith("/launch"):
        requests.post("http://control:8000/api/command", json={
            "command": "launch"

            output = run_agent(step["input"])

def track_cost(tokens, model):
    cost_per_m = {
        "gemma": 0.02,
        "gpt": 0.30
    }

    cost = (tokens / 1_000_000) * cost_per_m[model]

    r.incrbyfloat("aios_cost", cost)

    return cost

Login Controller
public function login(Request $request)
{
    if (!Auth::attempt($request->only('email', 'password'))) {
        return response()->json(['error' => 'Invalid credentials'], 401);
    }

    $user = Auth::user();
    $token = $user->createToken('auth_token')->plainTextToken;

    return response()->json([
        'token' => $token,
        'user' => $user
    ]);
}

Vue Auth Store (Pinia)
import { defineStore } from 'pinia'

export const useAuth = defineStore('auth', {
  state: () => ({
    user: null,
    token: localStorage.getItem('token')
  }),

  actions: {
    async login(email, password) {
      const res = await fetch('/api/login', {
        method: 'POST',
        body: JSON.stringify({ email, password })
      })

      const data = await res.json()
      this.token = data.token
      this.user = data.user

      localStorage.setItem('token', data.token)
    }
  }
})

public function pay(Request $request)
{
    $data = [
        'payment_options' => 'card, mobilemoney',
        'amount' => $request->amount,
        'email' => $request->email,
        'tx_ref' => uniqid(),
        'currency' => 'USD',
        'redirect_url' => route('payment.callback')
    ];

    return Flutterwave::initializePayment($data);
}
